# 학생 취업 수요 조사
while True : 
    stno = input("학번 : ")
    city = input("희망 지역 : ")
    job = input("희망 직종 : ")
    money = input("희망 연봉 : ")

    with open("job.txt", "a", encoding="utf-8") as f :
        mydata = stno + "-" + city + "-" + job + "-" + money 
        f.write(mydata + "\n")
        
    x = input("계속 ?(y/n)")
    if x == "n" :
        break


# 취업 수요조사 사용

with open("job.txt", "r", encoding="utf-8") as f :
    mydata = f.readlines()
    print(mydata)

city = input("지역명 : ")
money = input("희망연봉 : ")

for i in mydata :
    # print(i.strip())
    result = i.split("-")
    if (city == result[1]  and int(result[3]) <= int(money)) :
        print(f"학번 : {result[0]}, 희망지역 : {result[1]}, 희망연봉 : {result[3].strip()} ")
